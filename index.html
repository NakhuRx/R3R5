เชื่อม Google Apps Script

ข้อมูลเก็บในเครื่องเสมอ และส่งไป Sheet อัตโนมัติเมื่อกดบันทึกผู้ป่วย / จำนวนรายสัปดาห์ (สร้าง Google Sheet → Extensions → Apps Script → วางโค้ดด้านล่าง → Deploy เป็น Web app, Anyone)

function sh(n){var s=SpreadsheetApp.getActiveSpreadsheet();return s.getSheetByName(n)||s.insertSheet(n)}
function doPost(e){var d=JSON.parse(e.postData.contents),w=d.type=='day',s=sh(w?'Days':'Records'),k=(w?'d':'r')+d.key,v=s.getDataRange().getValues(),r=-1;
for(var i=0;i<v.length;i++)if(v[i][0]==k){r=i+1;break}
var x=w?[k,JSON.stringify(d.data),d.data.date,d.data.total]:[k,JSON.stringify(d.data),d.data.date,d.data.hn,d.data.inr,d.data.newDose];
if(r>0)s.getRange(r,1,1,x.length).setValues([x]);else s.appendRow(x);return ContentService.createTextOutput('ok')}
function doGet(e){var p=function(n){return sh(n).getDataRange().getValues().filter(function(x){return x[0]}).map(function(x){return JSON.parse(x[1])})};
return ContentService.createTextOutput(JSON.stringify({records:p('Records'),days:p('Days')})).setMimeType(ContentService.MimeType.JSON)}
